% Ragged-right setting for the content column that refuses short last lines
% ("orphans"): every line, including the last, may fall short of the margin
% by only a bounded amount, so TeX rebalances wrapped titles and author lists
% instead of leaving one or two words dangling. Word spaces never stretch,
% so only the right edge is ragged. No hyphenation.
\newcommand{\balancedrag}{%
  \setlength{\rightskip}{0pt plus 0.45\rightcol}%
  \setlength{\parfillskip}{0pt}%
  \spaceskip=\fontdimen2\font minus \fontdimen4\font
  \xspaceskip=\spaceskip
  \hbadness=10000 \hyphenpenalty=10000 \exhyphenpenalty=10000 \emergencystretch=0pt}

   \setlength{\rightcol}{\dimexpr\linewidth-\leftcol-\colsep\relax}%
   \noindent\begin{longtable}[l]{@{}>{\raggedleft\arraybackslash}p{\leftcol}@{\hspace{\colsep}}>{\raggedright\balancedrag\arraybackslash}p{\rightcol}@{}}
   % The heading is the table's first row, so it never strands at a page bottom
   \multicolumn{2}{@{}l}{\Large #1} \\*\noalign{\nobreak\vspace{0.4em}}}
  {\end{longtable}}
